rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null
        && request.auth.token.firebase.sign_in_provider == 'google.com'
        && request.auth.token.email_verified == true;
    }
    function activeEmployee() {
      return signedIn()
        && exists(/databases/$(database)/documents/employees/$(request.auth.uid))
        && get(/databases/$(database)/documents/employees/$(request.auth.uid)).data.active == true
        && get(/databases/$(database)/documents/employees/$(request.auth.uid)).data.email == request.auth.token.email;
    }
    match /employees/{uid} {
      allow get: if signedIn() && request.auth.uid == uid;
      allow list: if false;
      allow create, update, delete: if false;
    }
    // Shared app data: only active, registered employees can read/write.
    // Role-specific restrictions are still enforced by the app UI in this initial shared-document model.
    match /appData/{documentId} {
      allow read, write: if activeEmployee() && documentId == 'shared';
    }
    // Deny every collection not explicitly listed above.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
